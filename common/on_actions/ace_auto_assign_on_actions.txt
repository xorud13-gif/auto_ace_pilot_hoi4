on_actions = {
	# 1. 에이스 사망 (일반 적기에 격추)
	on_ace_killed = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:ACE_DIED"
			}
		}
	}

	# 2. 에이스 사망 (적 에이스에게 격추)
	on_ace_killed_by_ace = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:ACE_DIED"
			}
		}
	}

	# 3. 에이스 사망 (상호 격추)
	on_aces_killed_each_other = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:ACE_DIED"
			}
		}
	}

	# 4. 에이스 사망 (비행 사고)
	on_ace_killed_on_accident = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:ACE_DIED"
			}
		}
	}

	# 5. 신임 에이스 진급 (작전 중 새로운 에이스 등장 시 즉시 빈 비행단에 배치)

		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:PROMOTED"
			}
		}
	}

	# 6. 매일 자정 메인 스레드 펄스 (시간 진행 중 매일 1회 정기 검사 및 자동 배치)
	on_daily = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:DAILY"
			}
		}
	}
}
