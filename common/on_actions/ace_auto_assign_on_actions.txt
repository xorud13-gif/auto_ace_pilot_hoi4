on_actions = {
	# 1. 에이스 사망 (일반 적기에 격추)
	on_ace_killed = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:ACE_DIED"
			}
		}
	}

	# 2. 에이스 사망 (적 에이스에게 격추)
	on_ace_killed_by_ace = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:ACE_DIED"
			}
		}
	}

	# 3. 에이스 사망 (상호 격추)
	on_aces_killed_each_other = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:ACE_DIED"
			}
		}
	}

	# 4. 에이스 사망 (비행 사고)
	on_ace_killed_on_accident = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:ACE_DIED"
			}
		}
	}

	# 5. 신임 에이스 진급 (작전 중 새로운 에이스 등장 시 즉시 빈 비행단에 배치)
	on_ace_promoted = {
		effect = {
			if = {
				limit = {
					is_ai = no
					NOT = { has_country_flag = ace_auto_assign_disabled }
				}
				log = "ACE_AUTO_ASSIGN:PROMOTED"
			}
		}
	}

	# 6. 세이브 로드 / 게임 시작 시 비활성화 플래그 동기화
	on_startup = {
		effect = {
			if = {
				limit = {
					has_country_flag = ace_auto_assign_disabled
				}
				log = "ACE_AUTO_ASSIGN:DISABLE"
			}
		}
	}
}
